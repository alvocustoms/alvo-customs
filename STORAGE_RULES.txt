rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    function owner() { return request.auth != null && request.auth.token.email == 'alvo.customs@gmail.com'; }
    match /{allPaths=**} {
      allow read: if true;
      allow write: if owner() && request.resource.size < 25 * 1024 * 1024
        && request.resource.contentType.matches('image/.*|video/.*');
    }
  }
}
